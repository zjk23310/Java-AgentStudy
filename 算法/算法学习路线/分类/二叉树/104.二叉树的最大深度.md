```c++
class Solution {
public:
    
    int maxDepth(TreeNode* root) {
     if(root==nullptr)
        return 0;
     int Lref=maxDepth(root->left);
     int Rref=maxDepth(root->right);
         return max(Lref,Rref)+1;
    }
};
```
