我觉得这一题的递归还是很有趣的，主要是对在什么时候比较这块卡了一会
```c++
class Solution {
public:
    bool isSymmetric(TreeNode* root) {
        return root==nullptr?false:judge(root->left,root->right);
    }
    bool judge(TreeNode* roo1,TreeNode* roo2)
    {
        if(roo1==nullptr&&roo2==nullptr)
        return true;
        else if(roo1==nullptr||roo2==nullptr)
        return false;
        bool sL= judge(roo1->left,roo2->right);
        bool sR= judge(roo1->right,roo2->left);
        if(roo1->val==roo2->val)
        return sL&&sR;
        else
        return false;
    }
};
```
不过后面还是想到了怎么写的，当然，这一题并不是很难。

# 官解
```cpp
class Solution {
public:
    bool check(TreeNode *p, TreeNode *q) {
        if (!p && !q) return true;
        if (!p || !q) return false;
        return p->val == q->val && check(p->left, q->right) && check(p->right, q->left);
    }

    bool isSymmetric(TreeNode* root) {
        return check(root->left, root->right);
    }
};
```
