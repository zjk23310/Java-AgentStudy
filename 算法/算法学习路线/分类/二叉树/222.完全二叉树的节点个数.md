这个题目有一些很好的东西
首先如果一个完全没有规律的二叉树，要怎么才能得到节点个数呢？这里肯定要想到遍历。
那么我们知道遍历可以分为层序遍历和递归的dfs遍历。所以按照则这个思路，很明显的有两个解法
# 层序遍历解决没有规律二叉树个数
```cpp
class Solution {
public:
    int countNodes(TreeNode* root) {
        queue<TreeNode*> q;
        int ans=0;
        if(root==nullptr)
        return 0;
        q.push(root);
        int layer=0;
        while(!q.empty())
        {
            int size=q.size();
            if(1<<layer!=size)
            {
                ans+=size;
                return ans;
            }
            for(int i=0;i<size;++i)
            {
                TreeNode* top=q.front();
                q.pop();
                if(top->left!=nullptr)
                q.push(top->left);
                if(top->right!=nullptr)
                q.push(top->right);
            }
            ans+=size;
            layer++;
        }
        return ans;
    }
};
```

# 递归解决
```cpp
class Solution {
private:
    int getNodesNum(TreeNode* cur) {
        if (cur == NULL) return 0;
        int leftNum = getNodesNum(cur->left);      // 左
        int rightNum = getNodesNum(cur->right);    // 右
        int treeNum = leftNum + rightNum + 1;      // 中
        return treeNum;
    }
public:
    int countNodes(TreeNode* root) {
        return getNodesNum(root);
    }
};
```
更简化的写法可以是
```c++
class Solution {
public:
    int countNodes(TreeNode* root) {
        if (root == NULL) return 0;
        return 1 + countNodes(root->left) + countNodes(root->right);
    }
};
```
上面展示了两种方法。递归确实十分的优雅，而且感觉它的思想有种很高级的感觉
但是这两种方法在leetcode的这一题中都违反了规则（我记得之前leetcode不是只检查时间复杂度之类的吗，怎么开始检查是否违反限制了）
为什么呢？从题目就可以看出来了，这一题的二叉树是完全二叉树，完全二叉树与没有结构特性的二叉树相比是特殊的，所以用这种普遍的方法很明显会减低速度。那么我该怎么写呢？
不卖关子
```cpp
class Solution {
public:
    int countNodes(TreeNode* root) {
        if (root == nullptr) return 0;
        TreeNode* left = root->left;
        TreeNode* right = root->right;
        int leftDepth = 0, rightDepth = 0; // 这里初始为0是有目的的，为了下面求指数方便
        while (left) {  // 求左子树深度
            left = left->left;
            leftDepth++;
        }
        while (right) { // 求右子树深度
            right = right->right;
            rightDepth++;
        }
        if (leftDepth == rightDepth) {
            return (2 << leftDepth) - 1; // 注意(2<<1) 相当于2^2，所以leftDepth初始为0
        }
        return countNodes(root->left) + countNodes(root->right) + 1;
    }
};
```
这里求深度以及左子树深度等与右子树深度非常的有想法，这恰好用到了完全二叉树的性质，如果最左和最右都有节点则它是一个满树
