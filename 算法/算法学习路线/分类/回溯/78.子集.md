这一题回溯还是比较简单的，一开始想要使用for循环的回溯，但是理解错了，导致只有在index\=\=nums.size()的时候才加入结果集，后面转换了思想，考虑根据子集的定义分为放入和不放入两种情况，使用回溯完成了

# 初见
如图所示，这个初见就想到了回溯，因为很明显，每次的状态都是都可能的选择与不选择，然后按照回溯的模板写了一下算法，随后稍微调制一下就对了
```cpp
class Solution {
public:
    void recall(vector<vector<int>> &ans,int index,vector<int>&temp,vector<int>&nums)
    {
        if(index==nums.size())
        {
            ans.emplace_back(temp);
            return;
        }
        recall(ans,index+1,temp,nums);
        temp.emplace_back(nums[index]);
        recall(ans,index+1,temp,nums);
        temp.pop_back();
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> temp;
        recall(ans,0,temp,nums);
        return ans;
    }
};
```
# 官解
## 迭代法
```cpp
class Solution {
public:
    vector<int> t;
    vector<vector<int>> ans;

    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();
        for (int mask = 0; mask < (1 << n); ++mask) {
            t.clear();
            for (int i = 0; i < n; ++i) {
                if (mask & (1 << i)) {
                    t.push_back(nums[i]);
                }
            }
            ans.push_back(t);
        }
        return ans;
    }
};
```
如图所示，其实我觉得这个算法一般，但也算是一种思路

## 递归
实际上官解的递归就是回溯算法的实现
```cpp
class Solution {
public:
    vector<int> t;
    vector<vector<int>> ans;

    void dfs(int cur, vector<int>& nums) {
        if (cur == nums.size()) {
            ans.push_back(t);
            return;
        }
        t.push_back(nums[cur]);
        dfs(cur + 1, nums);
        t.pop_back();
        dfs(cur + 1, nums);
    }

    vector<vector<int>> subsets(vector<int>& nums) {
        dfs(0, nums);
        return ans;
    }
};
```
