```c++
class Solution {
public:
    vector<vector<int>> ans;
    vector<int> temp;
    vector<bool> use;

    vector<vector<int>> permute(vector<int>& nums) {
        ans.clear();
        temp.clear();
        use.assign(nums.size(), false);   // ✅ 重置
        backtracking(nums);
        return ans;
    }

    void backtracking(vector<int>& nums) {
        if (temp.size() == nums.size()) {   // 用 temp.size() 判断深度
            ans.push_back(temp);
            return;
        }
        for (int i = 0; i < (int)nums.size(); ++i) {
            if (use[i]) continue;           // 已经用过 → 跳过
            use[i] = true;
            temp.push_back(nums[i]);
            backtracking(nums);
            temp.pop_back();
            use[i] = false;                 // ✅ 回溯 use
        }
    }
};
```

class Solution {

public:
    vector<vector<int>> ans;
    void recall(vector<int>& nums, int index, vector<int> anstmp)
    {
        if (index == nums.size())
        {
            ans.push_back(anstmp);
            return;
        }
        //遍历插入不同位置
        for (int i = 0; i <= index; ++i)
        {
            anstmp.insert(anstmp.begin() + i, nums[index]);//不同位置插入第i个元素
            //继续调用回溯算法
            recall(nums, index + 1, anstmp);
            anstmp.erase(anstmp.begin() + i);
        }
    }
    
    vector<vector<int>> permute(vector<int>& nums) {
        vector<int> anstmp;
        recall(nums, 0, anstmp);
        return ans;
    }
};
