这一题也是非常之有趣的。分割回文串这一题给我的启发还挺大的，这一题也有点使用了[[131.分割回文串]]的思想，但是大头在如何处理不合规的数，主要采用了stoi转换为数字，然后比对数字，之后就是确保不会出现头为零的数字，具体代码如下
```c++
class Solution {
public:
    vector<string> ans;
    vector<string> restoreIpAddresses(string s) {
        string t;
        backtracking(s,0,0,t);
        return ans;
    }
    void backtracking(string& s,int index,int part,string &t)
    {
        if(part==4)
        { 
            if(t.size()==s.size()+3)
            ans.push_back(t);
            return;
        }
        string temp=t;
        or(int i=index;i<index+3&&i<s.size();++i)
        {
            string numString=s.substr(index,i-index+1);
            int num=stoi(numString);
            if(num>255||(numString[0]=='0'&&numString.size()>1))
            break;
            t+=s.substr(index,i-index+1);
            if(part<3)
            t+=".";
            backtracking(s,i+1,part+1,t);
            t=temp;
        }
    }
};
```
