对于回溯算法还是不够熟悉，不是很清楚模板该怎么使用，先贴一下我自己的代码：

```cpp
class Solution {
public:

    vector<int> dp;
    vector < vector<int>> ans;//存储答案
    int minx = INT_MAX;
    void recall(vector<int>& candidates, int target, int temp, vector<int> anstemp)
    {
        if (temp == target)
            ans.push_back(anstemp);
        if (temp+minx>target||dp[temp] != 0 )//该节点已访问，或者不可能存在答案
            return;

        for (int i = 0; i < candidates.size(); ++i)
        {
            int Itemp = candidates[i];
            anstemp.push_back(Itemp);
                 recall(candidates, target, temp + Itemp, anstemp);
            anstemp.pop_back();
        }
    }

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        dp.resize(target);
        //引入记忆化搜索
        for (int i = 0; i < candidates.size(); ++i)
            minx = min(minx, candidates[i]);
        for (int i = 0; i < target; ++i)
            dp[i] = 0;//表示未探索
        vector<int> anstemp;
        recall(candidates, target, 0, anstemp);
        return ans;
    }
};
```

问题：本来我是打算采用记忆化搜索的，但是实际上这样回溯好像不是很能用得上记忆化（后面我也意识到了不应该使用dp，不过答案还是存在重复的问题（这主要是因为我每次都从0索引开始进行遍历，实际上这会造成重复），答案也是使用回溯算法，题目中要求：一个数字可以重复，答案给出的算法是：

1.直接跳过，选择下一个数

2.减去这个数。修改组合数组（为了存储答案做准备）

``` cpp
class Solution1 {
public:

    vector < vector<int>> ans;

    void recall(vector<int>& candidates, int temp, vector<int> &anstemp,int index)
    {
        if (index == candidates.size())//表明所有元素都遍历过
            return;
        if (temp == 0)
        {
            ans.emplace_back(anstemp);//存放正确集合
            return;
        }
        //直接跳过操作
        recall(candidates, temp, anstemp, index + 1);

        //对下标元素进行操作
       if(temp-candidates[index]>=0){
            anstemp.emplace_back(candidates[index]);
            recall(candidates, temp - candidates[index], anstemp, index);
            anstemp.pop_back();
        }

    }

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        
        vector<int> anstemp;
```
