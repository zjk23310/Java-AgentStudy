# 初见
这一题想到一个解实际上并不是很难，困难点在于怎么优化......一开始使用了循环，果不其然就超时了，后面尝试减小计算，也就是如果max在开头，那么我就不需要循环了。只能说很接近答案了，而且答案的思想我实际上也想过。
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int i=0,j=k-1;
        int maxNum=INT_MIN;
        vector<int> ans;
         for( int temp=i;temp<=j;++temp)
                    {
                  maxNum=max(maxNum,nums[temp]);
                    }
        while(j<nums.size())
        {
            if(nums[i]==maxNum)
            {
                if(nums[j]>maxNum)
                {
                    maxNum=nums[j];
                ans.emplace_back(maxNum);
                ++i;
                ++j;
                }
                else
                {
                    maxNum=nums[i];
                    ans.emplace_back(maxNum);
                    maxNum=INT_MIN;
                    ++i;
                    ++j;
                    if(j==nums.size())
                    break;
                    for( int temp=i;temp<=j;++temp)
                    {
                  maxNum=max(maxNum,nums[temp]);
                    }
                }
            }    
            else{
                if(nums[j]>maxNum)
                {
                    maxNum=nums[j];
                }
                ans.emplace_back(maxNum);
                ++i;
                ++j;
            }
        }
        return ans;
    }
};
```
# 优先队列
可以参考[[优先队列]]
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        priority_queue<pair<int, int>> q;
        for (int i = 0; i < k; ++i) {
            q.emplace(nums[i], i);
        }
        vector<int> ans = {q.top().first};
        for (int i = k; i < n; ++i) {
            q.emplace(nums[i], i);
            while (q.top().second <= i - k) {
                q.pop();
            }
            ans.push_back(q.top().first);
        }
        return ans;
    }
};
```
这里面最关键的一步在于while()这一步，这一步的理解是如果我放入了一个值它比所有值都大，那么什么时候会出去呢？显然是当窗口的左边掠过它，这个处理非常有意思。
# 单调队列
这一题的思路主要是移除不必要的下标，思考这样一个事实：如果i\<j且nums\[i]\<nums\[j]，会发生什么？实际上这个时候nums\[i]是没必要存在的，如果我们将nums\[j]存入队列，我们可以将nums\[i]移出
可以进行这样操作：将每个nums\[j]放入队列前先进行比较，如果比队尾元素小，则放入队列，如果比队尾元素大则需要将队尾元素移出，然后放入
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        deque<int> q;
        for (int i = 0; i < k; ++i) {
            while (!q.empty() && nums[i] >= nums[q.back()]) {
                q.pop_back();
            }
            q.push_back(i);
        }

        vector<int> ans = {nums[q.front()]};
        for (int i = k; i < n; ++i) {
            while (!q.empty() && nums[i] >= nums[q.back()]) {
                q.pop_back();
            }
            q.push_back(i);
            while (q.front() <= i - k) {
                q.pop_front();
            }
            ans.push_back(nums[q.front()]);
        }
        return ans;
    }
};
```
[[C++常见的STL的使用方法]]中可以见到deque（双端队列）
这个对于求得区间内部最大值的方法实际上和上面差不多，对于头部元素的处理也与上面的方法相差不大
# 分块+预处理
将nums从左到右每k个进行分组，最后一组可能不足k个如果要求nums\[i]到nums\[i+k-1]的最大值有两种情况：
1.i是k的倍数，nums\[i]到nums\[i+k-1]是一个分组。预处理出每一个分组的最大值就好了
2.i不是k的倍数，nums\[i]到nums\[i+k-1]跨越两个分组，占有第一个分组的后缀还有第二个分组的前缀。如果我们能够预处理出每个分组中的前缀最大值以及后缀最大值，同样可以在 O(1) 的时间得到答案。
这一题可以参考[[稀疏表]]很像
```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> prefixMax(n), suffixMax(n);
        for (int i = 0; i < n; ++i) {
            if (i % k == 0) {
                prefixMax[i] = nums[i];
            }
            else {
                prefixMax[i] = max(prefixMax[i - 1], nums[i]);
            }
        }
        for (int i = n - 1; i >= 0; --i) {
            if (i == n - 1 || (i + 1) % k == 0) {
                suffixMax[i] = nums[i];
            }
            else {
                suffixMax[i] = max(suffixMax[i + 1], nums[i]);
            }
        }

        vector<int> ans;
        for (int i = 0; i <= n - k; ++i) {
            ans.push_back(max(suffixMax[i], prefixMax[i + k - 1]));
        }
        return ans;
    }
};
```
这个思路感觉很有趣，这种预处理的方式在解决问题前先处理了一遍，之后查询都是O(1)时间复杂度
