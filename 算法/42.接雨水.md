# 初见
这一题其实思考了差不多几分钟我就感觉可以使用单调栈，因为这种东西仿佛都是可以夹在两个单调点之间的，不过我遇到了几个问题，比如实际上他们有宽度属性，后面我尝试使用pair表示一对属性，该属性有高度，有宽度
然后我就开始尝试构建单调栈，并且着手写如何使用单调栈来解决这一个问题，一开始我没有考虑到，如果有体积就必须需要“左右两块都要有东西夹着”，右边好处理，因为是单调栈的条件，那么怎么考虑左边呢？实际上左边有两种情况可以讨论：
1.左边值小于右边值，则实际上如果栈为空则不需要进行下一步操作
2.左边值大于右边值，则循环特定循环会结束

## 我的代码
```cpp
class Solution {
public:
    int trap(vector<int>& height) {
        if (height.size() <= 2)//边界条件
            return 0;
        //6,4,2,0,3,2,0,3,1,4,5,3,2,7,5,3,0,1,2,1,3,4,6,8,1,3
        stack<pair<int, int>> s;//第一个参数高度、第二个宽度
        int maxHeight = height[0];
        int val = 0;
        //初始化栈
        s.push(pair<int, int>(height[0], 1));
        for (int i = 1; i < height.size(); ++i)
        {
            if (s.top().first > height[i])//高度问题
            {
                s.push(pair<int, int>(height[i], 1));//放入
            }
            else//栈不单调
            {
                int width = 1;//默认宽度
                pair<int, int> top;
                top = s.top();
                s.pop();
                if (top.first == height[i])//与前一个值相同
                {
                    top.second++;//宽度加一
                    s.push(top);
                    //不要考虑，继续循环
                    continue;
                }
                if (s.empty())//为空，说明开头没有东西挡着
                {
                    maxHeight = height[i];
                    s.push(pair<int,int>(height[i], 1));
                    //
                    continue;
                }
                int h = min(maxHeight, height[i]);
                maxHeight = max(maxHeight, height[i]);
                val += (h - top.first)*top.second;//初始
                width+=top.second;
                while (s.top().first <= height[i])//栈不空||top.first<hei...
                {
                    pair<int, int> st=s.top();//取值
                    s.pop();
                    if (s.empty())//直接返回
                        break;
                    val += (h - st.first)*st.second;//加体积
                    width+=st.second;
                }
                //入栈
                if(s.empty())
                    s.push(pair<int, int>(height[i], 1));
                else
                    s.push(pair<int, int>(height[i], width));
            }
        }
        return val;
    }
};
```
