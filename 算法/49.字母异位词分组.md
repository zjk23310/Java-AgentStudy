# 方法一：排序加哈希
采用排序加哈希的方法，由于字母异位词实际上排序后的结果是相同的，所以我们可以直接使用排序算法，并使用哈希表来记录之前是否存在这个结果
```cpp
class Solution {
public:
    //对字符串进行排序
    void sortString(string& s, int left, int right)//采用引用
 {
     if (left >= right)
         return;
     int pre = s[left];//基准
     int i = left - 1;
     int j = right + 1;
     while (i < j) {
         do i++; while (s[i] < pre);
         do j--; while (s[j] > pre);
         if(i<j)
           swap(s[i], s[j]);//交换
     }
     sortString(s, left, j);
     sortString(s, j + 1, right);
 }
 vector<vector<string>> groupAnagrams(vector<string>& strs) {
     //创建哈希
     vector<vector<string>> ans;
     unordered_map<string,int> map;
     int index=0;
     for(int i=0;i<strs.size();++i)
     {
        string temp=strs[i];
        sortString(temp,0,temp.size()-1);
        if(map.find(temp)!=map.end())//能找到
        ans[map[temp]].push_back(strs[i]);//放入
        else
        {
            map[temp]=index;
            index++;
            ans.resize(index);
            ans[index-1].push_back(strs[i]);
        }
     }
     return ans;
 }

};
```

# 方法二：计数：
主要是计算每个字母出现的次数，然后将所有出现过的字母以及字母出现的次数都拼接成一个字符串，之后依旧是使用哈希表进行判断是否出现过
![[49.字母异位词分组-1776341930000.webp]]
```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        // 自定义对 array<int, 26> 类型的哈希函数
        auto arrayHash = [fn = hash<int>{}] (const array<int, 26>& arr) -> size_t {
            return accumulate(arr.begin(), arr.end(), 0u, [&](size_t acc, int num) {
                return (acc << 1) ^ fn(num);
            });
        };

        unordered_map<array<int, 26>, vector<string>, decltype(arrayHash)> mp(0, arrayHash);
        for (string& str: strs) {
            array<int, 26> counts{};
            int length = str.length();
            for (int i = 0; i < length; ++i) {
                counts[str[i] - 'a'] ++;
            }
            mp[counts].emplace_back(str);
        }
        vector<vector<string>> ans;
        for (auto it = mp.begin(); it != mp.end(); ++it) {
            ans.emplace_back(it->second);
        }
        return ans;
    }
};
```
