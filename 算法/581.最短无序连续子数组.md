# 初见
这个初见一开始就思考了可不可以用单调栈来写，但是和遗憾没有成功，后面又想可不可以根据当前下标与该下标的前一个元素之间的关系来判断。但是后面算法越写越复杂，最后也就放弃了，不过后面还是有最后的绝招的，使用排序算法，然后从左右各找到一个与原排序不同的元素，这个还是比较直观且简单的。
```cpp
class Solution {
public:
    int findUnsortedSubarray(vector<int>& nums) {
        vector<int> temp = nums;
        sort(temp.begin(), temp.end());
        int left = 0, right = nums.size() - 1;
        bool Left = true, Right = true;
        while (left<nums.size()&&Left)
        {
            if (nums[left] != temp[left])
                {
                    Left = false;
                    break;
                }
            left++;
        }
        while (right>-1&&Right)
        {
            if (nums[right] != temp[right])
               {
                 Right = false;
                 break;
               }
            right--;
        }
        if (!Left && !Right)
            return right - left + 1;
        else
            return 0;
    }
};
```

```cpp
class Solution {
public:
    int findUnsortedSubarray(vector<int>& nums) {
        int n = nums.size();
        int maxn = INT_MIN, right = -1;
        int minn = INT_MAX, left = -1;
        for (int i = 0; i < n; i++) {
            if (maxn > nums[i]) {
                right = i;
            } else {
                maxn = nums[i];
            }
            if (minn < nums[n - i - 1]) {
                left = n - i - 1;
            } else {
                minn = nums[n - i - 1];
            }
        }
        return right == -1 ? 0 : right - left + 1;
    }
}
```
