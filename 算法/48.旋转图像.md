class Solution {
public:
        void rotate(vector<vector<int>>& matrix) {
        //只需要确定头位置的互换就好,以纵列为例,不过这不是非原地了
        int n = matrix.size();
        vector<vector<int> > rematrix(n,vector<int>(n));
        for (int i = 0; i < n; ++i)
        {
            for (int j = 0; j < n; ++j)
            {
                rematrix[j][n - i - 1] = matrix[i][j];
            }
        }
        for (int i = 0; i < n; ++i)
            for (int j = 0; j < n; ++j)
                matrix[i][j] = rematrix[i][j];
    }
};

# 方法二：原地操作
第二个主要是使用了问题进行分析找到了规律
假定：一开始的坐标是m[ row ] [ col ] 上面方法一也说了，实际上经过第一次变换之后m[col] [n-row-1] ,将该点的值存入temp，继续分析如果这个值改变，那么temp应该修改数组中哪个元素才能够保证不丢失数据，实际上再次经过操作存入应为m [n-cow-1] [n-row-1] 按照这个思路，最后会发现实际上，下次操作的目标就是m \[row] \[col]
  那么接下来的问题就是该对哪些元素进行操作（毕竟一次只能够确保四个元素得到循环，实际上，这个要分奇数偶数情况讨论

# 方法三：翻转代替旋转
先进行水平翻转，之后再进行对角线交换就可以实现顺时针旋转的效果
![[48.旋转图像-1776337258033.webp|112]]

``` cpp
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n = matrix.size();
        // 水平翻转
        for (int i = 0; i < n / 2; ++i) {
            for (int j = 0; j < n; ++j) {
                swap(matrix[i][j], matrix[n - i - 1][j]);
            }
        }
        // 主对角线翻转
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < i; ++j) {
                swap(matrix[i][j], matrix[j][i]);
            }
        }
    }
};
```
