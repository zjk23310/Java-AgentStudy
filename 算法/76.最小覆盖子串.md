    unordered_map <char, int> ori, cnt;

        for (const auto &p: ori) {

                return false;
            }
        }

    }

        while (r < int(s.size())) {
            if (ori.find(s[++r]) != ori.end()) {//如果该字符在要查询地集合中
                ++cnt[s[r]];
            }
            while (check() && l <= r) {//len为最小长度
                if (r - l + 1 < len) {//如果小于最小长度
                    len = r - l + 1;
                    ansL = l;//确定左边界
                }
                if (ori.find(s[l]) != ori.end()) {//如果在集合中则减去现在
                    --cnt[s[l]];
                }
                ++l;//移动左边界
            }
        }

        return ansL == -1 ? string() : s.substr(ansL, len);
    }
};

class Solution {
public:
    string minWindow(string s, string t) {
        const int n = s.size(), m = t.size();
        vector<int> mp1(128), mp2(128);
        for (auto ch : t) {
            mp1[ch]++;
        }
        int cnt = 0, best_i = -1, best_sz = n + 1, i = 0, j = -1;
        while (j < n) {
            if (cnt < m) {
                if (++j < n) {                    
                    if (++mp2[s[j]] <= mp1[s[j]]) {
                        cnt++;
                    }
                }
            } else {
                if (j - i + 1 < best_sz) {
                    best_sz = j - i + 1;
                    best_i = i;
                }
                if (--mp2[s[i]] < mp1[s[i]]) {
                    cnt--;
                }
                i++;
            }
        }
        if (best_i == -1) {
            return "";
        }
        return s.substr(best_i, best_sz);
    }
};

        int diff[128]{};

        for(char  c : t)
        {
            if(!diff[c])
                {
                    kinds++;
                }
            diff[c]--;
        }
        int m = s.size();
        int ansl=-1,ansr=m;
        int gecnt = 0;
        int left = 0;

            diff[c]++;
