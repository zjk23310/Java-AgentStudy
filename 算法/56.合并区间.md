# 初见：
这一题我一开始的算法实际上有些问题
```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        vector<vector<int>> ans;
        vector<vector<int>>  indexes(2*intervals.size());
        if (intervals.size() == 0)
            return ans;
        for (int i = 0; i < intervals.size(); ++i)
        {
            indexes[2 * i].push_back(intervals[i][0]);
            indexes[2 * i].push_back(0);//左区间
            indexes[2 * i + 1].push_back(intervals[i][1]);
            indexes[2 * i + 1].push_back(1);
        }//初始化
        sort(indexes.begin(), indexes.end());
        vector<int> temp(2);
        temp[0] = indexes[0][0];
        ans.push_back(temp);
        //已排序，那么进行第二步
        for (int i = 1; i < indexes.size(); ++i)
        {
            if (indexes[i][1] == 0)//表示左区间
            {
                if (i - 1 >= 0)
                    if (indexes[i - 1][0] != indexes[i][0] && indexes[i - 1][1] != 0)//上一个位置不是自己或者上一个位置不是左
                    {
                        vector<int> temp(2);
                        temp[0] = indexes[i][0];
                        ans.push_back(temp);
                    }
            }
            else
            {
                ans[ans.size() - 1][1] = indexes[i][0];//右区间的确定
            }
        }
        return ans;
    }
};
```
如上所示，是我的初版代码，实际上问题在于：我一开始想的是将区间转换成一个整的vector二维数组，然后根据是左区间还是右区间来对他们进行一个划分，但是这样做导致了一个问题：~={red}我打破了他们呢的内在逻辑，排完序之后我并不知道他们究竟是谁对应谁，这样就导致我无法正确地合并区间=~
那么这个时候就要想到另一种方式了
```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        vector<vector<int>> ans;
        sort(intervals.begin(), intervals.end());
        int right=INT_MIN;
        for(int i=0;i<intervals.size();++i)
        {
            if (intervals[i][0] > right)
            {
                ans.emplace_back(intervals[i]);//放入这个集合
                right = intervals[i][1];//右边集合
            }
            else {
                if (intervals[i][1] > right)//右边区间要大于现在右边区间
                {
                    ans[ans.size() - 1][1] = intervals[i][1];
                    right = intervals[i][1];
                }
            }
        }
        return ans;
    }
};
```
直接对原数组进行操作，并且创建一个辅助变量记录最右边的区间就可以了
