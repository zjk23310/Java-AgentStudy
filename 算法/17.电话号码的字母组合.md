class Solution {
public:
     char String[8][4] = {
    {'a','b','c'},
    {'d','e','f'},
    {'g','h','i'},
    {'j','k','l'},
    {'m','n','o'},
    {'p','q','r','s'},
    {'t','u','v'},
    {'w','x','y','z'},
     };
    vector<string> ans;
    void recall(string& s, int index, string temp)
    {
        if (index == s.size() )
        {
            ans.emplace_back(temp);
            return;
        }
        int j = 3;
        int num = s[index] - '2';
        if (num == 7 || num == 5)、
            j = 4;
        for (int i = 0; i < j; ++i)//三种情况'
        {
            char ch = String[num][i];
            recall(s, index + 1, temp + ch);
        }
    }
    vector<string> letterCombinations(string digits) {
        recall(digits, 0, "");
        return ans;
    }
};
