这一题第一个想法就是快速排序，后面发现自己理解错了，不过这一题可以使用二分查找找到相应的数据（这样可以向左，向右分别查询）
```cpp
class Solution {

public:

    vector<int> searchRange(vector<int>& nums, int target) {
        int left = 0, right = nums.size() - 1;
        int mid;
        vector<int> ans(2);
        ans[0] = -1;
        while (left <= right)
        {
            mid = (left + right) / 2;//中间元素
            if (nums[mid] < target)
                left = mid + 1;
            else if (nums[mid] > target)
                right = mid - 1;
            else
            {
               ans[0] = mid;
                break;
            }
        }
        if (ans[0] == -1)//根本就没有这个元素
            ans[1] = -1;
        else {
            int pre = 0, behind = 0;
            for (pre = mid; pre>-1&&nums[pre] == target; --pre);
            for (behind = mid; behind<nums.size()&&nums[behind] == target; ++behind)
            ans[0] = pre+1;
            ans[1] = behind-1;
        }
        return ans;
    }
};
```
这在最坏的情况下，时间复杂度会退化为O(n)，因此其实可以使用：另外一种思路：全程使用二分查找
```cpp
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int> ans(2, -1);
        if (nums.empty()) return ans;
        // 找左边界
        int left = 0, right = nums.size() - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] < target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }  
        // 检查是否找到
        if (left >= nums.size() || nums[left] != target) {
            return ans;
        }
        ans[0] = left;  
        // 找右边界
        right = nums.size() - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid] <= target) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        ans[1] = right;      
        return ans;
    }
};
```
