class Solution {
public:

    int minStep=INT_MAX;
    void recall(vector<vector<int>>&grid,int sum,int x,int y)
    {
        if(x==grid.size()-1&&y==grid[0].size()-1)
        {
            minStep=min(minStep,sum);
            return;
            }
        if(sum>minStep)//增加剪枝条件
        return;
        //到达边界判定
        if(x==grid.size()-1)
        {
            int temp=sum;
            for(y=y+1;y<grid[0].size();++y)
            {
                sum+=grid[x][y];
                if(sum>=minStep)
                return;
            }
            minStep=sum;
        }
        if(y==grid[0].size()-1)
        {
            int temp=sum;
            for(x=x+1;x<grid.size();++x)
            {
                sum+=grid[x][y];
                if(sum>=minStep)
                return;
            }
            minStep=sum;
        }
        //下或右两条路
        if(x+1<grid.size())
            recall(grid,sum+grid[x+1][y],x+1,y);
        if(y+1<grid[0].size())//未超过左边界
            recall(grid,sum+grid[x][y+1],x,y+1);

    }
    int minPathSum(vector<vector<int>>& grid) {
        recall(grid,grid[0][0],0,0);    
        return minStep;
    }
};

# 动态规划
```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int row = grid.size();
        int col = grid[0].size();
        vector<vector<int>>dp(row, vector<int>(col));
        for (int i = 0; i < row; ++i)
        {
            for (int j = 0; j < col; ++j)
            {
                if (i == 0)
                {
                    if (j == 0)
                        dp[i][j] = grid[i][j];
                    else dp[i][j] = grid[i][j]+dp[i][j-1];
                }
                else if (j == 0)
                {
                    if (i == 0)
                        dp[i][j] = grid[i][j];
                    else dp[i][j] = grid[i][j] + dp[i-1][j ];
                }
                else
                    dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
            }
        }
        return dp[row - 1][col - 1];
    }
};
```

```cpp
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        if (grid.size() == 0 || grid[0].size() == 0) {
            return 0;
        }
        int rows = grid.size(), columns = grid[0].size();
        auto dp = vector < vector <int> > (rows, vector <int> (columns));
        dp[0][0] = grid[0][0];
        for (int i = 1; i < rows; i++) {
            dp[i][0] = dp[i - 1][0] + grid[i][0];
        }
        for (int j = 1; j < columns; j++) {
            dp[0][j] = dp[0][j - 1] + grid[0][j];
        }
        for (int i = 1; i < rows; i++) {
            for (int j = 1; j < columns; j++) {
                dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];
            }
        }
        return dp[rows - 1][columns - 1];
    }
};
```
这时官解给的答案，我觉得我的时间效率低下的主要原因在于我的for循环中有if判断，导致每次for循环的时候都要进行if判断，从而拖慢了效率
