这一题也是初见过，但是问题在于怎么能够存储一层一层的元素，我使用了一个很笨的方法：创建了两个队列，并且对队列进行轮换操作，这其实大幅增加了空间复杂度，所以这个方法并不是很好，那么该怎么解决呢？


```cpp
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector <vector <int>> ret;
        if (!root) {
            return ret;
        }

        queue <TreeNode*> q;
        q.push(root);
        while (!q.empty()) {
            int currentLevelSize = q.size();
            ret.push_back(vector <int> ());
            for (int i = 1; i <= currentLevelSize; ++i) {
                auto node = q.front(); q.pop();
                ret.back().push_back(node->val);
                if (node->left) q.push(node->left);
                if (node->right) q.push(node->right);
            }
        }
        
        return ret;
    }
};
```
额，没错，答案就是使用一个temp变量记录每一层的元素个数就好了，内层再嵌入一个for循环。。。
```cpp
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        queue<TreeNode*> q,q1;
        int tempFlag=1;
        int layer = 0;
        q.push(root);
        vector<vector<int>>ans;
        while (!q.empty()||!q1.empty())
        {
            if (tempFlag)//进行一个轮换操作
            {
                vector<int> ansTemp;
                while (!q.empty())
                {
                    TreeNode* temp = q.front();
                    q.pop();
                    if (temp == nullptr)
                        continue;
                    else
                    {
                        ansTemp.push_back(temp->val);
                        q1.push(temp->left);
                        q1.push(temp->right);
                    }
                }
                if (ansTemp.size() != 0)
                    ans.emplace_back(ansTemp);
                tempFlag = 0;
            }
            else {
                vector<int> ansTemp;
                while (!q1.empty())
                {
                    TreeNode* temp = q1.front();
                    q1.pop();
                    if (temp == nullptr)
                        continue;
                    else
                    {
                        ansTemp.push_back(temp->val);
                        q.push(temp->left);
                        q.push(temp->right);
                    }
                }
                if (ansTemp.size() != 0)
                    ans.emplace_back(ansTemp);
                tempFlag = 1;
            }
        }
        return ans;
    }
};
```
