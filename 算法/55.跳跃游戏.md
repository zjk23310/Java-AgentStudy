```cpp
class Solution {
public:

    bool canJump(vector<int>& nums) {
    if(nums.size()<=1)
    return true;
    int distanse=0;
    for(int i=0;i<nums.size();++i)
    {
        if(i==distanse&&nums[i]==0)
        return false;
        if(i+nums[i]>distanse)
        {
            distanse=i+nums[i];
            if(distanse>=nums.size()-1)
            return true;
        }
    }
    return false;
    }
};
```
